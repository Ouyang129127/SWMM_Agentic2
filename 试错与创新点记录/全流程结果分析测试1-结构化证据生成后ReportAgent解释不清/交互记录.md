import json
from pathlib import Path

def extract_cell_claim_info(claims_file_path, cell_id, output_file):
    """
    提取指定cell的claim详细信息，特别是引用的证据ID

    Args:
        claims_file_path: diagnosis_claims.json文件路径
        cell_id: 要查询的cell ID
        output_file: 输出文件路径
    """

    # 检查文件是否存在
    claims_path = Path(claims_file_path)
    if not claims_path.exists():
        raise FileNotFoundError(f"诊断claims文件不存在: {claims_path}")

    # 读取JSON文件
    with open(claims_path, 'r', encoding='utf-8') as f:
        claims_data = json.load(f)

    # 查找指定cell的claim
    cell_claims = []
    for claim in claims_data.get('claims', []):
        if claim.get('cell_id') == cell_id:
            cell_claims.append(claim)

    # 写入输出文件
    output_path = Path(output_file)
    output_path.parent.mkdir(parents=True, exist_ok=True)

    with open(output_path, 'w', encoding='utf-8') as f:
        f.write(f"Cell {cell_id} 的Claim详细信息\n")
        f.write("=" * 50 + "\n\n")

        if not cell_claims:
            f.write(f"未找到cell {cell_id} 的claim信息\n")
            return

        for i, claim in enumerate(cell_claims, 1):
            f.write(f"Claim {i}:\n")
            f.write(f"  Claim ID: {claim.get('claim_id', 'N/A')}\n")
            f.write(f"  Cell ID: {claim.get('cell_id', 'N/A')}\n")
            f.write(f"  Claim类型: {claim.get('claim_type', 'N/A')}\n")
            f.write(f"  Claim描述: {claim.get('description', 'N/A')}\n")
            f.write(f"  严重程度: {claim.get('severity', 'N/A')}\n")
            f.write(f"  置信度: {claim.get('confidence', 'N/A')}\n")

            # 提取证据ID
            evidence_ids = claim.get('evidence_ids', [])
            f.write(f"  引用的证据ID ({len(evidence_ids)}个):\n")
            for evidence_id in evidence_ids:
                f.write(f"    - {evidence_id}\n")

            # 其他可能的相关信息
            if 'related_claims' in claim:
                f.write(f"  相关Claim IDs: {claim.get('related_claims', [])}\n")

            if 'timestamp' in claim:
                f.write(f"  时间戳: {claim.get('timestamp', 'N/A')}\n")

            f.write("\n" + "-" * 40 + "\n\n")

    print(f"Cell {cell_id} 的claim信息已保存到: {output_path}")
