import json
from pathlib import Path

def list_rainfall_events(project_root="models/songhua_swmm_2d"):
    """List all rainfall event files in the events/ directory."""
    events_dir = Path(project_root) / "events"

    if not events_dir.exists():
        raise ValueError(f"Events directory not found: {events_dir}")

    rainfall_files = []

    # Look for common rainfall file extensions
    rainfall_extensions = {'.csv', '.txt', '.dat', '.rain', '.precip'}

    for file_path in events_dir.rglob('*'):
        if file_path.is_file() and file_path.suffix.lower() in rainfall_extensions:
            rainfall_files.append(file_path)

    # Also check for JSON event files that might contain rainfall data
    for file_path in events_dir.rglob('*.json'):
        if file_path.is_file():
            try:
                with open(file_path, 'r', encoding='utf-8') as f:
                    data = json.load(f)
                    # Check if it's a rainfall event file by looking for common keys
                    if any(key in data for key in ['rainfall', 'precipitation', 'timeseries', 'event_data']):
                        rainfall_files.append(file_path)
            except (json.JSONDecodeError, UnicodeDecodeError):
                continue

    return sorted(rainfall_files)

def save_rainfall_events_list(output_file="rainfall_events_list.txt"):
    """Save the list of rainfall events to a text file."""
    try:
        rainfall_events = list_rainfall_events()

        with open(output_file, 'w', encoding='utf-8') as f:
            f.write("Rainfall Events in songhua_swmm_2d Project\n")
            f.write("=" * 50 + "\n\n")

            if not rainfall_events:
                f.write("No rainfall event files found in events/ directory.\n")
            else:
                f.write(f"Found {len(rainfall_events)} rainfall event file(s):\n\n")
                for i, event_file in enumerate(rainfall_events, 1):
                    f.write(f"{i}. {event_file}\n")
                    f.write(f"   Relative path: {event_file.relative_to(Path('models/songhua_swmm_2d'))}\n")
                    f.write(f"   File size: {event_file.stat().st_size} bytes\n\n")

        print(f"Rainfall events list saved to: {output_file}")

        # Print summary to console
        print(f"\nSummary: Found {len(rainfall_events)} rainfall event file(s)")
        for event_file in rainfall_events:
            print(f"  - {event_file.relative_to(Path('models/songhua_swmm_2d'))}")

    except Exception as e:
        print(f"Error: {e}")
        raise

import matplotlib.pyplot as plt
import pandas as pd
from datetime import datetime
from pathlib import Path

def plot_rainfall_hyetograph():
    """Create a hyetograph plot of the rainfall event."""

    # Read the rainfall event file
    rainfall_file = Path("models/songhua_swmm_2d/events/rain1.txt")

    if not rainfall_file.exists():
        raise FileNotFoundError(f"Rainfall file not found: {rainfall_file}")

    # Read and parse the data
    rainfall_data = []
    with open(rainfall_file, 'r', encoding='utf-8') as f:
        for line in f:
            line = line.strip()
            if line and "," in line and not line.startswith("#") and "单位" not in line:
                try:
                    datetime_str, rainfall_str = line.split(",")
                    rainfall_value = float(rainfall_str)
                    dt = datetime.strptime(datetime_str.strip(), "%Y-%m-%d %H:%M:%S")
                    rainfall_data.append((dt, rainfall_value))
                except ValueError:
                    continue

    if not rainfall_data:
        print("No valid rainfall data found")
        return

    # Create DataFrame for easier plotting
    df = pd.DataFrame(rainfall_data, columns=['datetime', 'rainfall_mm_h'])

    # Create the plot
    plt.figure(figsize=(12, 6))

    # Plot hyetograph (rainfall intensity over time)
    plt.bar(df['datetime'], df['rainfall_mm_h'], width=0.02,
            color='blue', alpha=0.7, edgecolor='darkblue')

    # Customize the plot
    plt.title('Rainfall Hyetograph - rain1.txt', fontsize=14, fontweight='bold')
    plt.xlabel('Time', fontsize=12)
    plt.ylabel('Rainfall Intensity (mm/h)', fontsize=12)
    plt.grid(True, alpha=0.3)

    # Format x-axis
    plt.xticks(rotation=45)
    plt.tight_layout()

    # Add statistics annotation
    max_rainfall = df['rainfall_mm_h'].max()
    total_rainfall = df['rainfall_mm_h'].sum() * 0.5  # 30-minute intervals
    duration_hours = len(df) * 0.5

    stats_text = f"Max: {max_rainfall} mm/h\nTotal: {total_rainfall:.1f} mm\nDuration: {duration_hours} hrs"
    plt.annotate(stats_text, xy=(0.02, 0.95), xycoords='axes fraction',
                bbox=dict(boxstyle="round,pad=0.3", fc="white", alpha=0.8),
                fontsize=10, verticalalignment='top')

    # Save the plot
    output_dir = Path("outputs")
    output_dir.mkdir(exist_ok=True)

    plot_file = output_dir / "rainfall_hyetograph.png"
    plt.savefig(plot_file, dpi=300, bbox_inches='tight')
    plt.show()

    print(f"Rainfall hyetograph saved to: {plot_file}")
    print(f"Event duration: {duration_hours} hours")
    print(f"Peak intensity: {max_rainfall} mm/h")
    print(f"Total rainfall: {total_rainfall:.1f} mm")

def get_file_info(file_path):
    """获取文件大小和基本信息"""
    stat = file_path.stat()

    # 尝试读取文件内容获取基本信息
    info = f"Size: {size_kb:.2f} KB"

    # 根据文件扩展名提供特定信息
    if file_path.suffix.lower() in ['.csv', '.txt', '.dat']:
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                first_line = f.readline().strip()
                if first_line:
                    info += f", First line: {first_line[:100]}"
        except:
            info += ", Could not read content"

    return info

def inventory_rainfall_events(project_root="models/songhua_swmm_2d"):
    """列出降雨事件文件清单"""
    events_dir = Path(project_root) / "events"

    if not events_dir.exists():
        raise FileNotFoundError(f"Events directory not found: {events_dir}")

    rainfall_files = []

    # 递归查找所有文件
    for file_path in events_dir.rglob("*"):
        if file_path.is_file():
            # 排除隐藏文件和目录
            if not file_path.name.startswith('.'):
                rainfall_files.append(file_path)

    # 按路径排序
    rainfall_files.sort()

    return rainfall_files, events_dir
